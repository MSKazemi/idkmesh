#!/usr/bin/env python3
"""Validate IDKMesh benchmark cohort indexes and their existing evidence contracts.

The cohort is an index, not a new task/execution protocol. WorkUnit, EvaluatorPlan,
ResultManifest, and VerificationResult remain canonical. This tool adds the
cross-object checks needed to freeze a reusable benchmark without allowing task
labels, evaluator bindings, or structural-signature taxonomy to drift after
outcomes are known.
"""

from __future__ import annotations

import argparse
import copy
import hashlib
import json
from pathlib import Path, PurePosixPath
import re
import sys
from typing import Any

from jsonschema import Draft202012Validator, FormatChecker

ROOT = Path(__file__).resolve().parents[1]
SCHEMA_DIR = ROOT / "schemas"
COHORT_SCHEMA = SCHEMA_DIR / "benchmark-cohort-v0.1.schema.json"
WORK_UNIT_SCHEMA = SCHEMA_DIR / "work-unit-v0.2.schema.json"
EVALUATOR_PLAN_SCHEMA = SCHEMA_DIR / "evaluator-plan-v0.2.schema.json"
RESULT_MANIFEST_SCHEMA = SCHEMA_DIR / "result-manifest-v0.1.schema.json"
VERIFICATION_RESULT_SCHEMA = SCHEMA_DIR / "verification-result-v0.1.schema.json"
FULL_GIT_SHA = re.compile(r"^[0-9a-f]{40}$")

RECOMMENDATION_TO_OUTCOME = {
    "accept_candidate": "support",
    "reject_candidate": "reject",
    "escalate": "escalate",
    "insufficient_evidence": "insufficient_evidence",
}
VERIFICATION_FINDING_CATEGORIES = {
    "correctness",
    "scope",
    "provenance",
    "security",
    "regression",
}


class CohortError(RuntimeError):
    """Raised when a benchmark cohort or referenced evidence fails closed."""


def require(condition: bool, message: str) -> None:
    if not condition:
        raise CohortError(message)


def canonical_digest(value: Any) -> str:
    payload = json.dumps(
        value,
        sort_keys=True,
        separators=(",", ":"),
        ensure_ascii=False,
    ).encode("utf-8")
    return "sha256:" + hashlib.sha256(payload).hexdigest()


def load_json(path: Path) -> dict[str, Any]:
    try:
        value = json.loads(path.read_text(encoding="utf-8"))
    except (OSError, json.JSONDecodeError) as exc:
        raise CohortError(f"cannot load JSON from {path}: {exc}") from exc
    if not isinstance(value, dict):
        raise CohortError(f"expected JSON object in {path}")
    return value


def _schema(path: Path) -> dict[str, Any]:
    schema = load_json(path)
    Draft202012Validator.check_schema(schema)
    return schema


def validate_schema(instance: dict[str, Any], path: Path, label: str) -> None:
    validator = Draft202012Validator(_schema(path), format_checker=FormatChecker())
    errors = sorted(validator.iter_errors(instance), key=lambda error: list(error.absolute_path))
    if not errors:
        return
    details: list[str] = []
    for error in errors[:12]:
        location = ".".join(str(part) for part in error.absolute_path) or "<root>"
        details.append(f"{location}: {error.message}")
    raise CohortError(f"{label} failed schema validation: " + "; ".join(details))


def resolve_repo_file(raw: str, *, label: str) -> Path:
    posix = PurePosixPath(raw)
    require(bool(raw) and not posix.is_absolute() and ".." not in posix.parts, f"{label}: unsafe path {raw!r}")
    path = (ROOT / Path(*posix.parts)).resolve()
    try:
        path.relative_to(ROOT)
    except ValueError as exc:
        raise CohortError(f"{label}: path escapes repository root: {raw}") from exc
    require(path.is_file(), f"{label}: referenced file does not exist: {raw}")
    require(not path.is_symlink(), f"{label}: symlink references are not allowed: {raw}")
    return path


def definition_projection(cohort: dict[str, Any]) -> dict[str, Any]:
    """Return the pre-outcome benchmark definition committed by definition_digest.

    Evidence/results, lifecycle stage, storage paths, and extensions are excluded.
    The digest therefore survives result attachment and file relocation while still
    committing task identity, source snapshot, taxonomy, evaluator digest, negative
    expectation, accounting requirements, and predeclared structural signatures.
    """

    tasks: list[dict[str, Any]] = []
    for task in sorted(cohort["tasks"], key=lambda item: item["id"]):
        evaluator = task["evaluator"]
        tasks.append(
            {
                "id": task["id"],
                "family": task["family"],
                "difficulty": task["difficulty"],
                "split": task["split"],
                "source": {
                    "repository": task["source"]["repository"],
                    "revision": task["source"]["revision"],
                },
                "work_unit": {
                    "id": task["work_unit"]["id"],
                    "version": task["work_unit"]["version"],
                    "digest": task["work_unit"]["digest"],
                },
                "evaluator": {
                    "visibility": evaluator["visibility"],
                    "plan_id": evaluator["plan_id"],
                    "plan_digest": evaluator["plan_digest"],
                    "backend": evaluator["backend"],
                },
                "declared_structural_signatures": sorted(task["declared_structural_signatures"]),
                "negative_case": {
                    "description": task["negative_case"]["description"],
                    "expected_category": task["negative_case"]["expected_category"],
                },
                "accounting": {
                    "required_metrics": sorted(task["accounting"]["required_metrics"]),
                },
            }
        )
    return {
        "schema_version": cohort["schema_version"],
        "id": cohort["id"],
        "title": cohort["title"],
        "minimum_final_tasks": cohort["minimum_final_tasks"],
        "required_families": sorted(cohort["required_families"]),
        "taxonomy_frozen_before_outcomes": cohort["taxonomy_frozen_before_outcomes"],
        "authority": cohort["authority"],
        "tasks": tasks,
    }


def definition_digest(cohort: dict[str, Any]) -> str:
    return canonical_digest(definition_projection(cohort))


def _required_validator_ids(work_unit: dict[str, Any]) -> set[str]:
    return {
        item["id"]
        for item in work_unit["validators"]
        if item.get("required") is True
    }


def _load_and_match_object_ref(ref: dict[str, Any], *, schema: Path, label: str) -> dict[str, Any]:
    path = resolve_repo_file(ref["path"], label=label)
    value = load_json(path)
    validate_schema(value, schema, label)
    require(canonical_digest(value) == ref["digest"], f"{label}: canonical digest drift")
    require(value.get("id") == ref["id"], f"{label}: id drift")
    return value


def _validate_public_plan(task: dict[str, Any], work_unit: dict[str, Any]) -> dict[str, Any] | None:
    evaluator = task["evaluator"]
    if evaluator["visibility"] == "hidden":
        require("plan_path" not in evaluator, f"{task['id']}: hidden evaluator must not expose plan_path")
        return None

    plan_path = resolve_repo_file(evaluator["plan_path"], label=f"{task['id']} EvaluatorPlan")
    plan = load_json(plan_path)
    validate_schema(plan, EVALUATOR_PLAN_SCHEMA, f"{task['id']} EvaluatorPlan")
    require(plan["id"] == evaluator["plan_id"], f"{task['id']}: EvaluatorPlan id drift")
    require(canonical_digest(plan) == evaluator["plan_digest"], f"{task['id']}: EvaluatorPlan digest drift")
    require(plan["backend"]["type"] == evaluator["backend"], f"{task['id']}: evaluator backend drift")

    binding = plan["binding"]
    require(binding["work_unit_id"] == work_unit["id"], f"{task['id']}: EvaluatorPlan WorkUnit id drift")
    require(binding["work_unit_version"] == work_unit["version"], f"{task['id']}: EvaluatorPlan WorkUnit version drift")
    require(binding["work_unit_digest"] == canonical_digest(work_unit), f"{task['id']}: EvaluatorPlan WorkUnit digest drift")
    require(binding["source_revision"] == task["source"]["revision"], f"{task['id']}: EvaluatorPlan source revision drift")
    require(
        set(plan["required_validator_ids"]) == _required_validator_ids(work_unit),
        f"{task['id']}: EvaluatorPlan required-validator set drift",
    )
    return plan


def _validate_attempt(
    *,
    task: dict[str, Any],
    work_unit: dict[str, Any],
    attempt: dict[str, Any],
) -> None:
    require(
        attempt["structural_signature"] in task["declared_structural_signatures"],
        f"{task['id']}/{attempt['attempt_id']}: structural signature was not predeclared",
    )
    result = _load_and_match_object_ref(
        attempt["result_manifest"],
        schema=RESULT_MANIFEST_SCHEMA,
        label=f"{task['id']}/{attempt['attempt_id']} ResultManifest",
    )
    verification = _load_and_match_object_ref(
        attempt["verification_result"],
        schema=VERIFICATION_RESULT_SCHEMA,
        label=f"{task['id']}/{attempt['attempt_id']} VerificationResult",
    )

    work_unit_digest = canonical_digest(work_unit)
    require(result["work_unit_id"] == work_unit["id"], f"{task['id']}: ResultManifest WorkUnit id drift")
    require(result["work_unit_version"] == work_unit["version"], f"{task['id']}: ResultManifest WorkUnit version drift")
    require(result["provenance"]["work_unit_digest"] == work_unit_digest, f"{task['id']}: ResultManifest WorkUnit digest drift")
    require(result["provenance"]["source_revision"] == task["source"]["revision"], f"{task['id']}: ResultManifest source revision drift")
    require(
        _required_validator_ids(work_unit).issubset(
            set(result["verification_request"]["expected_validator_ids"])
        ),
        f"{task['id']}: ResultManifest omitted required validator request",
    )

    result_digest = canonical_digest(result)
    require(verification["result_manifest_id"] == result["id"], f"{task['id']}: VerificationResult ResultManifest id drift")
    require(verification["work_unit_id"] == work_unit["id"], f"{task['id']}: VerificationResult WorkUnit id drift")
    require(verification["work_unit_version"] == work_unit["version"], f"{task['id']}: VerificationResult WorkUnit version drift")
    require(verification["provenance"]["work_unit_digest"] == work_unit_digest, f"{task['id']}: VerificationResult WorkUnit digest drift")
    require(verification["provenance"]["result_manifest_digest"] == result_digest, f"{task['id']}: VerificationResult ResultManifest digest drift")
    require(verification["provenance"]["source_revision"] == task["source"]["revision"], f"{task['id']}: VerificationResult source revision drift")
    require(
        verification["provenance"].get("verifier_config_digest") == task["evaluator"]["plan_digest"],
        f"{task['id']}: VerificationResult is not bound to indexed EvaluatorPlan digest",
    )
    require(
        verification["independence"]["independent_from_worker"] is True,
        f"{task['id']}: analyzed verification is not independent from worker",
    )
    expected_outcome = RECOMMENDATION_TO_OUTCOME[verification["decision_support"]["recommendation"]]
    require(attempt["outcome"] == expected_outcome, f"{task['id']}: indexed attempt outcome drift")

    for metric in task["accounting"]["required_metrics"]:
        available = metric in result.get("resources", {}) or metric in verification.get("resources", {})
        require(available, f"{task['id']}/{attempt['attempt_id']}: missing required accounting metric {metric}")


def _validate_negative_case(task: dict[str, Any]) -> None:
    negative = task["negative_case"]
    if negative["evidence_status"] != "verified":
        return
    path = resolve_repo_file(negative["evidence_path"], label=f"{task['id']} negative evidence")
    value = load_json(path)
    require(canonical_digest(value) == negative["evidence_digest"], f"{task['id']}: negative-evidence digest drift")

    if negative["evidence_type"] == "verification_result":
        validate_schema(value, VERIFICATION_RESULT_SCHEMA, f"{task['id']} negative VerificationResult")
        require(
            value["decision_support"]["recommendation"] != "accept_candidate",
            f"{task['id']}: negative VerificationResult unexpectedly recommends acceptance",
        )
        expected = negative["expected_category"]
        if expected in VERIFICATION_FINDING_CATEGORIES:
            categories = {finding["category"] for finding in value["findings"]}
            require(expected in categories, f"{task['id']}: negative evidence lacks expected {expected} finding")


def validate_cohort(cohort: dict[str, Any], *, require_evidence: bool = False) -> dict[str, Any]:
    validate_schema(cohort, COHORT_SCHEMA, "BenchmarkCohort")

    task_ids = [task["id"] for task in cohort["tasks"]]
    require(len(task_ids) == len(set(task_ids)), "benchmark task ids must be unique")

    work_keys: set[tuple[str, str, str, int]] = set()
    for task in cohort["tasks"]:
        if cohort["stage"] != "scaffold":
            require(
                FULL_GIT_SHA.fullmatch(task["source"]["revision"]) is not None,
                f"{task['id']}: collecting/frozen benchmark source revision must be a full 40-hex Git SHA",
            )

        work_unit_ref = task["work_unit"]
        work_unit = _load_and_match_object_ref(
            work_unit_ref,
            schema=WORK_UNIT_SCHEMA,
            label=f"{task['id']} WorkUnit",
        )
        require(work_unit["version"] == work_unit_ref["version"], f"{task['id']}: WorkUnit version drift")
        require(
            work_unit["provenance"].get("source_revision") == task["source"]["revision"],
            f"{task['id']}: WorkUnit source revision drift",
        )

        work_key = (
            task["source"]["repository"],
            task["source"]["revision"],
            work_unit["id"],
            work_unit["version"],
        )
        require(work_key not in work_keys, f"{task['id']}: duplicate source/WorkUnit benchmark item")
        work_keys.add(work_key)

        _validate_public_plan(task, work_unit)
        attempt_ids = [attempt["attempt_id"] for attempt in task["evidence"]["attempts"]]
        require(len(attempt_ids) == len(set(attempt_ids)), f"{task['id']}: attempt ids must be unique within a task")
        if task["evidence"]["status"] == "verified":
            for attempt in task["evidence"]["attempts"]:
                _validate_attempt(task=task, work_unit=work_unit, attempt=attempt)
        elif task["evidence"]["attempts"]:
            raise CohortError(f"{task['id']}: pending/excluded task may not attach analyzed attempts")

        _validate_negative_case(task)
        if require_evidence and task["evidence"]["status"] == "pending":
            raise CohortError(f"{task['id']}: evidence is still pending")
        if require_evidence and task["evidence"]["status"] == "verified":
            require(
                task["negative_case"]["evidence_status"] == "verified",
                f"{task['id']}: included benchmark item lacks verified seeded-negative evidence",
            )

    if cohort["stage"] in {"frozen", "burned"}:
        require(
            len(cohort["tasks"]) >= cohort["minimum_final_tasks"],
            "frozen/burned cohort is smaller than minimum_final_tasks",
        )
        families = {task["family"] for task in cohort["tasks"]}
        require(
            set(cohort["required_families"]).issubset(families),
            "frozen/burned cohort does not cover all required task families",
        )

    expected_definition_digest = definition_digest(cohort)
    if "definition_digest" in cohort:
        require(
            cohort["definition_digest"] == expected_definition_digest,
            "benchmark pre-outcome definition digest drift",
        )

    return {
        "cohort_id": cohort["id"],
        "stage": cohort["stage"],
        "task_count": len(cohort["tasks"]),
        "definition_digest": expected_definition_digest,
        "verified_tasks": sum(task["evidence"]["status"] == "verified" for task in cohort["tasks"]),
        "excluded_tasks": sum(task["evidence"]["status"] == "excluded" for task in cohort["tasks"]),
        "pending_tasks": sum(task["evidence"]["status"] == "pending" for task in cohort["tasks"]),
        "automatic_candidate_selection": cohort["authority"]["automatic_candidate_selection"],
    }

def cmd_validate(args: argparse.Namespace) -> int:
    cohort = load_json((ROOT / args.cohort).resolve())
    summary = validate_cohort(cohort, require_evidence=args.require_evidence)
    print(json.dumps(summary, indent=2, sort_keys=True))
    return 0


def cmd_definition_digest(args: argparse.Namespace) -> int:
    cohort = load_json((ROOT / args.cohort).resolve())
    validate_schema(cohort, COHORT_SCHEMA, "BenchmarkCohort")
    print(definition_digest(cohort))
    return 0


def cmd_self_test(_: argparse.Namespace) -> int:
    baseline = _fixture_cohort()
    summary = validate_cohort(baseline)
    require(summary["pending_tasks"] == 1, "self-test scaffold was not classified as pending")

    original_digest = definition_digest(baseline)
    outcome_only = copy.deepcopy(baseline)
    outcome_only["tasks"][0]["evidence"] = {
        "status": "excluded",
        "attempts": [],
        "exclusion_reason": "self-test outcome mutation",
    }
    require(
        definition_digest(outcome_only) == original_digest,
        "attaching/changing outcomes changed the pre-outcome definition digest",
    )

    taxonomy_drift = copy.deepcopy(baseline)
    taxonomy_drift["tasks"][0]["difficulty"] = "small"
    require(
        definition_digest(taxonomy_drift) != original_digest,
        "taxonomy drift did not change the pre-outcome definition digest",
    )

    bad_work_digest = copy.deepcopy(baseline)
    bad_work_digest["tasks"][0]["work_unit"]["digest"] = "sha256:" + "0" * 64
    try:
        validate_cohort(bad_work_digest)
    except CohortError:
        pass
    else:
        raise CohortError("self-test expected WorkUnit digest drift to fail closed")

    bad_plan_digest = copy.deepcopy(baseline)
    bad_plan_digest["tasks"][0]["evaluator"]["plan_digest"] = "sha256:" + "0" * 64
    try:
        validate_cohort(bad_plan_digest)
    except CohortError:
        pass
    else:
        raise CohortError("self-test expected EvaluatorPlan digest drift to fail closed")

    collecting_fixture_revision = copy.deepcopy(baseline)
    collecting_fixture_revision["stage"] = "collecting"
    try:
        validate_cohort(collecting_fixture_revision)
    except CohortError:
        pass
    else:
        raise CohortError("self-test expected non-immutable collecting source revision to fail closed")

    print(
        "OK: benchmark cohort schema, cross-object WorkUnit/EvaluatorPlan binding, "
        "pre-outcome definition commitment, and fail-closed drift checks passed"
    )
    return 0


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(description=__doc__)
    subparsers = parser.add_subparsers(dest="command", required=True)

    validate = subparsers.add_parser("validate", help="Validate a benchmark cohort and referenced evidence.")
    validate.add_argument("--cohort", required=True, help="Repository-relative BenchmarkCohort JSON path.")
    validate.add_argument(
        "--require-evidence",
        action="store_true",
        help="Reject pending tasks and require verified seeded-negative evidence for included tasks.",
    )
    validate.set_defaults(func=cmd_validate)

    digest = subparsers.add_parser("definition-digest", help="Print the pre-outcome benchmark definition digest.")
    digest.add_argument("--cohort", required=True)
    digest.set_defaults(func=cmd_definition_digest)

    self_test = subparsers.add_parser("self-test", help="Run deterministic contract and drift tests.")
    self_test.set_defaults(func=cmd_self_test)
    return parser


def main() -> int:
    args = build_parser().parse_args()
    try:
        return args.func(args)
    except (CohortError, KeyError, TypeError, ValueError, OSError, json.JSONDecodeError) as exc:
        print(f"ERROR: {exc}", file=sys.stderr)
        return 2


if __name__ == "__main__":
    raise SystemExit(main())
